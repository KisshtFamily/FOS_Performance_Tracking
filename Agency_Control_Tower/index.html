function doGet(e) { var action = e.parameter.action; // 'summary' ya 'paid_file' var agency = e.parameter.agency; // 'ALL', 'ALPHA', 'APEX', etc. var metric = e.parameter.metric || 'COUNT'; // 'COUNT', 'POW', 'ROI' var ss = SpreadsheetApp.getActiveSpreadsheet(); if (action === "summary") { // Sirf dashboard ki summary table ka aggregated JSON return karega (No heavy raw rows) var summarySheet = ss.getSheetByName("Dashboard_Summary"); var data = summarySheet.getDataRange().getValues(); return ContentService.createTextOutput(JSON.stringify({ status: "success", agency: agency, metric: metric, data: data })).setMimeType(ContentService.MimeType.JSON); } if (action === "paid_file") { // Sirf jab user "Paid File" par click karega, tabhi direct CSV format me download hoga var rawSheet = ss.getSheetByName("Raw_Paid_Data"); var rawData = rawSheet.getDataRange().getValues(); // Agency filter condition var filtered = rawData.filter(function(row, idx) { if (idx === 0) return true; // headers if (agency === "ALL") return true; return row[1] === agency; // Column B: Agency ID }); var csvContent = filtered.map(function(r){ return r.join(","); }).join("\n"); return ContentService.createTextOutput(csvContent).setMimeType(ContentService.MimeType.CSV); } }
